# i18n con traduzione LLM — Piano di implementazione

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Aggiungere un sistema i18n con 30+ lingue, dove le traduzioni sono generate da Claude API su richiesta admin e salvate in Supabase — senza file di traduzione manuali.

**Architecture:** `next-intl` gestisce il runtime (nessun prefisso URL). Le stringhe UI sorgente vivono in `messages/source.json`; le traduzioni generate dal LLM stanno nella tabella `translations`. Il locale viene letto dal profilo utente (`users.locale`) e iniettato via `NextIntlClientProvider` nel root layout. Le label menu vengono tradotte al momento del fetch in `getMenuItems(locale)`.

**Tech Stack:** next-intl, @anthropic-ai/sdk, Supabase, Next.js 15 App Router, TypeScript

## Global Constraints

- App Router only — no Pages Router
- Tutti i comandi Node vanno eseguiti da `apps/web/`
- Tutti i comandi Python/pytest vanno eseguiti con `uv run` dalla root
- Il modello Claude per traduzioni: `claude-haiku-4-5-20251001`
- Source locale sentinel: stringa `'source'` (non un codice lingua reale)
- Le migrazioni SQL vanno applicate via MCP supabase (`mcp__supabase__apply_migration`) E aggiunte a `deploy/db/schema.sql`
- Nessun prefisso URL (`/en/...`) — locale dal profilo utente
- Fallback: se una chiave manca in DB, mostrare il valore da `source.json`

```ts
export function nestedToFlat(
  obj: Record<string, unknown>,
  prefix = ''
): Array<{ key: string; value: string }> {
  const result: Array<{ key: string; value: string }> = []
  for (const [k, v] of Object.entries(obj)) {
    const key = prefix ? `${prefix}.${k}` : k
    if (typeof v === 'string') {
      result.push({ key, value: v })
    } else if (v && typeof v === 'object' && !Array.isArray(v)) {
      result.push(...nestedToFlat(v as Record<string, unknown>, key))
    }
  }
  return result
}

export function flatKeysToNested(
  rows: Array<{ key: string; value: string }>
): Record<string, unknown> {
  const result: Record<string, unknown> = {}
  for (const { key, value } of rows) {
    const parts = key.split('.')
    let cur = result
    for (let i = 0; i < parts.length - 1; i++) {
      if (!cur[parts[i]] || typeof cur[parts[i]] !== 'object') {
        cur[parts[i]] = {}
      }
      cur = cur[parts[i]] as Record<string, unknown>
    }
    cur[parts[parts.length - 1]] = value
  }
  return result
}

export function deepMerge(
  base: Record<string, unknown>,
  override: Record<string, unknown>
): Record<string, unknown> {
  const result = { ...base }
  for (const [k, v] of Object.entries(override)) {
    if (
      v &&
      typeof v === 'object' &&
      !Array.isArray(v) &&
      result[k] &&
      typeof result[k] === 'object'
    ) {
      result[k] = deepMerge(
        result[k] as Record<string, unknown>,
        v as Record<string, unknown>
      )
    } else {
      result[k] = v
    }
  }
  return result
}
```

```ts
import { cache } from 'react'
import { createAdminClient } from '@/lib/supabase-server'
import { flatKeysToNested, deepMerge } from '@/lib/i18n-utils'

export const getUserLocale = cache(async (userId: string): Promise<string> => {
  const supabase = createAdminClient()
  const { data } = await supabase
    .from('users')
    .select('locale')
    .eq('id', userId)
    .single()
  return data?.locale ?? 'source'
})

export async function loadMessages(locale: string): Promise<Record<string, unknown>> {
  const source = (await import('@/messages/source.json')).default as Record<string, unknown>
  if (!locale || locale === 'source') return source

  const supabase = createAdminClient()
  const { data } = await supabase
    .from('translations')
    .select('key, value')
    .eq('locale', locale)
    .eq('namespace', 'ui')

  if (!data?.length) return source

  const dbNested = flatKeysToNested(data as Array<{ key: string; value: string }>)
  return deepMerge(source, dbNested)
}
```

```ts
'use server'

import { auth } from '@/lib/auth'
import { createAdminClient } from '@/lib/supabase-server'

export interface UserProfile {
  first_name: string | null
  last_name: string | null
  username: string | null
  phone: string | null
  locale: string | null
}

export interface LocaleOption {
  code: string
  name: string
  native_name: string
  flag_emoji: string | null
}

export async function getActiveLocales(): Promise<LocaleOption[]> {
  const supabase = createAdminClient()
  const { data } = await supabase
    .from('locales')
    .select('code, name, native_name, flag_emoji')
    .eq('active', true)
    .order('sort_order')
  return (data ?? []) as LocaleOption[]
}

export async function saveProfile(profile: UserProfile): Promise<{ error: string | null }> {
  const session = await auth()
  if (!session?.user?.id) return { error: 'Not authenticated' }

  const supabase = createAdminClient()
  const { error } = await supabase.from('users').upsert(
    {
      id: session.user.id,
      first_name: profile.first_name,
      last_name: profile.last_name,
      username: profile.username,
      phone: profile.phone,
      locale: profile.locale || null,
      updated_at: new Date().toISOString(),
    },
    { onConflict: 'id' }
  )

  return { error: error?.message ?? null }
}
```

```ts
export const getMenuItems = cache(async (locale?: string): Promise<MenuItem[]> => {
  const supabase = createAdminClient()

    .from('menu_items')
    .select('*')
    .order('order')

  if (error) throw new Error(`Failed to load menu: ${error.message}`)
  const items = (data ?? []).map(mapFromDb)

  if (!locale || locale === 'source') return items

  const keys = items.map(i => `item.${i.id}.label`)
  const { data: translations } = await supabase
    .from('translations')
    .select('key, value')
    .eq('locale', locale)
    .eq('namespace', 'menu')
    .in('key', keys)

  if (!translations?.length) return items

  const trMap: Record<string, string> = {}
  for (const t of translations) trMap[t.key] = t.value

  return items.map(item => ({
    ...item,
    localizedLabel: trMap[`item.${item.id}.label`],
  }))
})
```

```ts
import { NextRequest, NextResponse } from 'next/server'
import Anthropic from '@anthropic-ai/sdk'
import { auth } from '@/lib/auth'
import { createAdminClient } from '@/lib/supabase-server'
import { nestedToFlat } from '@/lib/i18n-utils'

const CHUNK_SIZE = 80

interface SourceString {
  namespace: string
  key: string
  value: string
}

interface TranslateRequest {
  locales: string[]
  overwrite: boolean
}

async function collectSourceStrings(): Promise<SourceString[]> {
  const supabase = createAdminClient()

  // UI strings from source.json
  const sourceJson = (await import('@/messages/source.json')).default as Record<string, unknown>
  const uiStrings = nestedToFlat(sourceJson).map(({ key, value }) => ({
    namespace: 'ui',
    key,
    value,
  }))

  // Menu labels
  const { data: menuItems } = await supabase
    .from('menu_items')
    .select('id, label')
  const menuStrings = (menuItems ?? []).map((item: { id: string; label: string }) => ({
    namespace: 'menu',
    key: `item.${item.id}.label`,
    value: item.label,
  }))

  return [...uiStrings, ...menuStrings]
}

async function translateChunk(
  strings: SourceString[],
  targetLocales: string[],
  anthropic: Anthropic
): Promise<{ locale: string; key: string; namespace: string; value: string }[]> {
  const inputMap: Record<string, string> = {}
  for (const s of strings) inputMap[`${s.namespace}::${s.key}`] = s.value

  const prompt = `Translate the following strings to these locales: ${targetLocales.join(', ')}.
Auto-detect the source language.
Respond ONLY with valid JSON, no extra text.

Response format:
{
  "en": { "ui::nav.save": "Save", "menu::item.abc.label": "Dashboard" },
  "fr": { "ui::nav.save": "Enregistrer", "menu::item.abc.label": "Tableau de bord" }
}

Strings to translate:
${JSON.stringify(inputMap, null, 2)}`

  const message = await anthropic.messages.create({
    model: 'claude-haiku-4-5-20251001',
    max_tokens: 8096,
    messages: [{ role: 'user', content: prompt }],
  })

  const text = message.content[0].type === 'text' ? message.content[0].text : ''

  // Strip markdown code fences if present
  const cleaned = text.replace(/^```(?:json)?\n?/m, '').replace(/\n?```$/m, '').trim()
  const parsed = JSON.parse(cleaned) as Record<string, Record<string, string>>

  const results: { locale: string; key: string; namespace: string; value: string }[] = []
  for (const [locale, translations] of Object.entries(parsed)) {
    for (const [compoundKey, value] of Object.entries(translations)) {
      const [namespace, ...keyParts] = compoundKey.split('::')
      results.push({ locale, namespace, key: keyParts.join('::'), value })
    }
  }
  return results
}

export async function POST(req: NextRequest) {
  const session = await auth()
  if (session?.user?.role !== 'admin') {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 403 })
  }

  const body = await req.json().catch(() => null) as TranslateRequest | null
  if (!body?.locales?.length) {
    return NextResponse.json({ error: 'locales array required' }, { status: 400 })
  }

  const { locales, overwrite } = body
  const supabase = createAdminClient()
  const anthropic = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY })

  let allStrings = await collectSourceStrings()

  // Filter out already-translated strings if not overwriting
  if (!overwrite) {
    const { data: existing } = await supabase
      .from('translations')
      .select('namespace, key, locale')
      .in('locale', locales)

    const existingSet = new Set(
      (existing ?? []).map((r: { namespace: string; key: string; locale: string }) =>
        `${r.namespace}::${r.key}::${r.locale}`
      )
    )

    allStrings = allStrings.filter(s =>
      locales.some(locale => !existingSet.has(`${s.namespace}::${s.key}::${locale}`))
    )
  }

  const skipped = (await collectSourceStrings()).length * locales.length - allStrings.length * locales.length
  const errors: string[] = []
  let translated = 0

  // Process in chunks
  for (let i = 0; i < allStrings.length; i += CHUNK_SIZE) {
    const chunk = allStrings.slice(i, i + CHUNK_SIZE)
    try {
      const results = await translateChunk(chunk, locales, anthropic)
      if (results.length > 0) {
        const rows = results.map(r => ({
          namespace: r.namespace,
          key: r.key,
          locale: r.locale,
          value: r.value,
          source_text: chunk.find(s => s.key === r.key && s.namespace === r.namespace)?.value ?? '',
          generated_at: new Date().toISOString(),
        }))
        const { error } = await supabase
          .from('translations')
          .upsert(rows, { onConflict: 'namespace,key,locale' })
        if (error) {
          errors.push(`Chunk ${i / CHUNK_SIZE + 1} DB error: ${error.message}`)
        } else {
          translated += results.length
        }
      }
    } catch (err) {
      // Retry once
      try {
        const results = await translateChunk(chunk, locales, anthropic)
        const rows = results.map(r => ({
          namespace: r.namespace,
          key: r.key,
          locale: r.locale,
          value: r.value,
          source_text: chunk.find(s => s.key === r.key && s.namespace === r.namespace)?.value ?? '',
          generated_at: new Date().toISOString(),
        }))
        await supabase.from('translations').upsert(rows, { onConflict: 'namespace,key,locale' })
        translated += results.length
      } catch (retryErr) {
        errors.push(`Chunk ${i / CHUNK_SIZE + 1}: ${retryErr instanceof Error ? retryErr.message : 'unknown error'}`)
      }
    }
  }

  return NextResponse.json({ translated, skipped, locales, errors })
}
```

```ts
export async function getCoverage(): Promise<CoverageRow[]> {
  const supabase = createAdminClient()

  // Count total source strings (UI + menu)
  const [{ count: uiTotal }, { count: menuTotal }] = await Promise.all([

    .from('locales')

    .order('sort_order')

  const results: CoverageRow[] = []
  for (const loc of locales ?? []) {
    const { count } = await supabase
      .from('translations')
      .select('*', { count: 'exact', head: true })
      .eq('locale', loc.code)
    results.push({
      code: loc.code,
      name: loc.name,
      native_name: loc.native_name,
      flag_emoji: loc.flag_emoji,
      translated: count ?? 0,
      total,
    })
  }
  return results
}
```
